cmake_minimum_required(VERSION 3.20)
project(Plasma LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
 message(FATAL_ERROR "Plasma requires a Windows x86 compiler")
endif()
include(cmake/Dependencies.cmake)
option(PLASMA_ENABLE_INTERNET "HTTP: update checks and downloads, the online manual" OFF)
option(PLASMA_ENABLE_LICENSING "The license VM (otherwise sheet.plx is loaded from the program directory)" OFF)
file(GLOB APP_SOURCES CONFIGURE_DEPENDS src/*.cpp support/*.cpp)
if(NOT PLASMA_ENABLE_INTERNET)
 list(FILTER APP_SOURCES EXCLUDE REGEX "/HttpClient\\.cpp$")
 list(APPEND APP_SOURCES optional/OfflineHttpClient.cpp)
endif()
add_executable(plasma WIN32 ${APP_SOURCES} platform/Wintab.cpp platform/FreeImageRuntime.cpp)
target_include_directories(plasma PRIVATE include)
target_compile_definitions(plasma PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS GLEW_STATIC FREEIMAGE_LIB)
if(NOT PLASMA_ENABLE_INTERNET)
 target_compile_definitions(plasma PRIVATE PLASMA_NO_INTERNET)
endif()
if(NOT PLASMA_ENABLE_LICENSING)
 target_compile_definitions(plasma PRIVATE PLASMA_NO_LICENSING)
endif()
target_link_libraries(plasma PRIVATE lcms jpeg freetype pugixml zlibstatic glew FreeImage kernel32 user32 gdi32 comdlg32 shell32 ole32 oleaut32 winmm ws2_32 mscms opengl32 glu32 wbemuuid)
if(MSVC)
 target_compile_options(plasma PRIVATE /std:c++17 /FICompiler.h /EHsc /bigobj)
else()
 target_compile_options(plasma PRIVATE -include Compiler.h -fwrapv -Wno-register)
 target_link_options(plasma PRIVATE -static -static-libgcc -static-libstdc++)
endif()
add_custom_command(TARGET plasma POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/assets/sheet.plx" "$<TARGET_FILE_DIR:plasma>/sheet.plx")
